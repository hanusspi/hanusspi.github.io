<script src="https://cdnjs.cloudflare.com/ajax/libs/Chart.js/3.9.1/chart.min.js"></script>
    
    <script>
        // Generate normal distribution data
        function normalPDF(x, mean, std) {
            return (1 / (std * Math.sqrt(2 * Math.PI))) * 
                   Math.exp(-0.5 * Math.pow((x - mean) / std, 2));
        }
        
        // X values from -1 to 1
        const x = [];
        for (let i = -1; i <= 1; i += 0.02) {
            x.push(i);
        }
        
        // Different normal distributions
        const curves = [
            { mean: 0, std: 0.2, color: 'orange', label: 'σ=0.2' },
            { mean: 0, std: 0.3, color: 'lightgreen', label: 'σ=0.3' },
            { mean: 0, std: 0.4, color: 'cyan', label: 'σ=0.4' },
            { mean: 0, std: 0.5, color: 'blue', label: 'σ=0.5' },
            { mean: 0, std: 0.6, color: 'purple', label: 'σ=0.6' }
        ];
        
        const datasets = curves.map(curve => ({
            label: curve.label,
            data: x.map(xi => ({ x: xi, y: normalPDF(xi, curve.mean, curve.std) })),
            borderColor: curve.color,
            backgroundColor: 'transparent',
            borderWidth: 3,
            pointRadius: 0,
            tension: 0.4
        }));
        
        // Add vertical line at x=0
        datasets.push({
            label: 'x=0',
            data: [{ x: 0, y: 0 }, { x: 0, y: 2.2 }],
            borderColor: 'red',
            backgroundColor: 'transparent',
            borderWidth: 2,
            pointRadius: 0,
            showLine: true
        });
        
        const ctx = document.getElementById('chart').getContext('2d');
        new Chart(ctx, {
            type: 'line',
            data: { datasets },
            options: {
                responsive: true,
                scales: {
                    x: {
                        type: 'linear',
                        min: -1,
                        max: 1,
                        ticks: { stepSize: 0.5 }
                    },
                    y: {
                        min: 0,
                        max: 2.5,
                        ticks: { stepSize: 0.5 }
                    }
                },
                plugins: {
                    legend: { display: false }
                }
            }
        });
    </script>
